## 🎨 DNIInputComponent

```typescript
// src/app/shared/components/dni-input/dni-input.component.ts
import { Component, Input, Output, EventEmitter } from '@angular/core';

@Component({
  selector: 'app-dni-input',
  template: `
    <div>
      <label *ngIf="label" class="block text-sm mb-1">{{ label }}</label>
      <input
        type="text"
        [value]="value"
        (input)="onInputChange($event)"
        maxlength="8"
        pattern="[0-9]*"
        inputmode="numeric"
        [placeholder]="placeholder"
        class="w-full p-3 border rounded-lg"
        [class.border-red-500]="error">
      <p *ngIf="error" class="text-sm text-red-500 mt-1">{{ error }}</p>
    </div>
  `
})
export class DNIInputComponent {
  @Input() value = '';
  @Input() label = 'DNI';
  @Input() placeholder = '12345678';
  @Input() error = '';
  @Output() valueChange = new EventEmitter<string>();

  onInputChange(event: Event): void {
    const input = event.target as HTMLInputElement;
    const value = input.value.replace(/\D/g, '').slice(0, 8);
    input.value = value;
    this.valueChange.emit(value);
  }
}
```

## 🔢 NumericStepperComponent

```typescript
// src/app/shared/components/numeric-stepper/numeric-stepper.component.ts
import { Component, Input, Output, EventEmitter } from '@angular/core';

@Component({
  selector: 'app-numeric-stepper',
  template: `
    <div class="flex items-center justify-center gap-4">
      <button
        (click)="decrement()"
        [disabled]="value <= min"
        class="w-12 h-12 bg-gray-100 rounded-lg disabled:opacity-50 hover:bg-gray-200 transition-colors">
        -
      </button>
      <span class="text-2xl w-12 text-center">{{ value }}</span>
      <button
        (click)="increment()"
        [disabled]="value >= max"
        class="w-12 h-12 bg-gray-100 rounded-lg disabled:opacity-50 hover:bg-gray-200 transition-colors">
        +
      </button>
    </div>
  `
})
export class NumericStepperComponent {
  @Input() value = 0;
  @Input() min = 0;
  @Input() max = 99;
  @Output() valueChange = new EventEmitter<number>();

  increment(): void {
    if (this.value < this.max) {
      this.value++;
      this.valueChange.emit(this.value);
    }
  }

  decrement(): void {
    if (this.value > this.min) {
      this.value--;
      this.valueChange.emit(this.value);
    }
  }
}
```

## 📋 TransactionListComponent

```typescript
// src/app/shared/components/transaction-list/transaction-list.component.ts
import { Component, Input } from '@angular/core';
import { Transaction } from '../../../core/models/transaction.model';

@Component({
  selector: 'app-transaction-list',
  template: `
    <div class="space-y-3">
      <div
        *ngFor="let transaction of transactions"
        class="bg-white rounded-lg shadow p-4 flex items-center gap-4">
        <div
          [class.bg-green-100]="transaction.type === 'recarga'"
          [class.bg-blue-100]="transaction.type === 'pasaje'"
          [class.bg-gray-100]="transaction.type === 'vinculacion'"
          class="w-12 h-12 rounded-full flex items-center justify-center flex-shrink-0">
          <span *ngIf="transaction.type === 'recarga'">💰</span>
          <span *ngIf="transaction.type === 'pasaje'">🚌</span>
          <span *ngIf="transaction.type === 'vinculacion'">🔗</span>
        </div>
        <div class="flex-1">
          <p class="mb-1">{{ transaction.description }}</p>
          <p *ngIf="transaction.route" class="text-xs text-gray-500">{{ transaction.route }}</p>
          <p class="text-xs text-gray-400">{{ formatDate(transaction.date) }}</p>
        </div>
        <div class="text-right">
          <p
            [class.text-green-600]="transaction.type === 'recarga'"
            [class.text-red-600]="transaction.type === 'pasaje'"
            class="mb-1">
            {{ transaction.type === 'recarga' ? '+' : '-' }}S/ {{ transaction.amount.toFixed(2) }}
          </p>
        </div>
      </div>
    </div>
  `
})
export class TransactionListComponent {
  @Input() transactions: Transaction[] = [];

  formatDate(date: Date): string {
    // Usar date-fns o formato manual
    const now = new Date();
    const diff = now.getTime() - new Date(date).getTime();
    const hours = Math.floor(diff / (1000 * 60 * 60));
    
    if (hours < 1) return 'Hace unos minutos';
    if (hours < 24) return `Hace ${hours} horas`;
    if (hours < 48) return 'Ayer';
    return new Date(date).toLocaleDateString('es-PE');
  }
}
```

```typescript
// src/app/shared/components/amount-selector/amount-selector.component.ts
import { Component, Input, Output, EventEmitter } from '@angular/core';

@Component({
  selector: 'app-amount-selector',
  template: `
    <div class="grid grid-cols-2 gap-3">
      <button
        *ngFor="let amount of amounts"
        (click)="selectAmount(amount)"
        [class.border-[#2563EB]]="selectedAmount === amount"
        [class.bg-blue-50]="selectedAmount === amount"
        class="p-4 border-2 rounded-lg hover:bg-gray-50 transition-colors">
        S/ {{ amount }}
      </button>
    </div>
  `
})
export class AmountSelectorComponent {
  @Input() amounts: number[] = [5, 10, 20, 50];
  @Input() selectedAmount: number | null = null;
  @Output() amountChange = new EventEmitter<number>();

  selectAmount(amount: number): void {
    this.selectedAmount = amount;
    this.amountChange.emit(amount);
  }
}
```

```typescript
import { Component, Input, Output, EventEmitter } from '@angular/core';

@Component({
  selector: 'app-map-controls',
  template: `
    <!-- Zoom controls -->
    <div class="absolute top-20 right-4 z-10 space-y-2">

        (click)="zoomIn.emit()"

      </button>
      <button
        (click)="zoomOut.emit()"

        -
      </button>
    </div>

        (click)="toggleFollowMode.emit()"
        [class.bg-[#2563EB]]="isFollowMode"

      </button>
    </div>
  `

  @Input() isLocating = false;
```
