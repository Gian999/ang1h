## 💰 COMPONENTE: RECHARGE

### recharge.component.ts
```typescript
import { Component } from '@angular/core';
import { Router } from '@angular/router';
import { TransactionService } from '../../../../core/services/transaction.service';

@Component({
  selector: 'app-recharge',
  templateUrl: './recharge.component.html'
})
export class RechargeComponent {
  selectedAmount: number | null = null;
  customAmount = '';
  selectedMethod: 'yape' | 'plin' | 'tarjeta' | 'agente' = 'yape';
  currentBalance = 0;

  readonly amounts = [5, 10, 20, 50];

  constructor(
    private transactionService: TransactionService,
    private router: Router
  ) {
    this.currentBalance = this.transactionService.calculateBalance();
  }

  selectAmount(amount: number) {
    this.selectedAmount = amount;
    this.customAmount = '';
  }

  onCustomAmountChange(value: string) {
    this.customAmount = value;
    this.selectedAmount = null;
  }

  get finalAmount(): number {
    return this.selectedAmount || parseFloat(this.customAmount) || 0;
  }

  onRecharge() {
    if (this.finalAmount <= 0) return;

    this.transactionService.addRecharge(this.finalAmount, this.selectedMethod);
    alert(`Recarga exitosa de S/ ${this.finalAmount.toFixed(2)}`);
    this.router.navigate(['/card']);
  }

  onBack() {
    this.router.navigate(['/card']);
  }
}
```

  </div>
</div>

## 📋 COMPONENTE: MOVEMENTS

### movements.component.ts
```typescript
import { Component, OnInit } from '@angular/core';
import { Router } from '@angular/router';
import { TransactionService } from '../../../../core/services/transaction.service';
import { Transaction } from '../../../../core/models/transaction.model';

@Component({
  selector: 'app-movements',
  templateUrl: './movements.component.html'
})
export class MovementsComponent implements OnInit {
  transactions: Transaction[] = [];
  filter: 'all' | 'recarga' | 'pasaje' = 'all';

  constructor(
    private transactionService: TransactionService,
    private router: Router
  ) {}

  ngOnInit() {
    this.loadTransactions();
  }

  loadTransactions() {
    if (this.filter === 'all') {
      this.transactions = this.transactionService.getTransactions();
    } else {
      this.transactions = this.transactionService.getTransactionsByType(this.filter);
    }
  }

  setFilter(filter: 'all' | 'recarga' | 'pasaje') {
    this.filter = filter;
    this.loadTransactions();
  }

  onBack() {
    this.router.navigate(['/card']);
  }
}
```

### map-view.component.ts
```typescript
import { Component, OnInit, OnDestroy, AfterViewInit } from '@angular/core';
import { Router } from '@angular/router';
import mapboxgl from 'mapbox-gl';

@Component({
  selector: 'app-map-view',
  templateUrl: './map-view.component.html',
  styleUrls: ['./map-view.component.css']
})
export class MapViewComponent implements OnInit, AfterViewInit, OnDestroy {
  private map!: mapboxgl.Map;
  private userMarker?: mapboxgl.Marker;
  private watchId?: number;
  
  isLocating = false;
  isFollowMode = false;
  userPosition: { lat: number; lng: number } | null = null;

  readonly MAPBOX_TOKEN = 'pk.eyJ1IjoibHVjay1wYXN0b3IiLCJhIjoiY21oOXhjczdsMDBkZjJybjFpam9oMG1qbCJ9.0Q8otsUabn_2ZQVouvfiJw';
  readonly JULIACA_CENTER: [number, number] = [-70.13, -15.5];

  constructor(private router: Router) {}

  ngOnInit() {
    this.getUserLocation();
  }

  ngAfterViewInit() {
    this.initMap();
  }

  ngOnDestroy() {
    if (this.watchId) {
      navigator.geolocation.clearWatch(this.watchId);
    }
    if (this.map) {
      this.map.remove();
    }
  }

  initMap() {
    mapboxgl.accessToken = this.MAPBOX_TOKEN;

    this.map = new mapboxgl.Map({
      container: 'map',
      style: 'mapbox://styles/mapbox/streets-v12',
      center: this.JULIACA_CENTER,
      zoom: 15,
      pitch: 0,      // Vista 2D
      maxPitch: 0    // Bloquear rotación 3D
    });

    this.map.addControl(new mapboxgl.NavigationControl());
  }

  getUserLocation() {
    if (!navigator.geolocation) {
      alert('Tu navegador no soporta geolocalización');
      return;
    }

    this.isLocating = true;

    navigator.geolocation.getCurrentPosition(
      (position) => {
        const pos = {
          lat: position.coords.latitude,
          lng: position.coords.longitude
        };
        this.userPosition = pos;
        this.isLocating = false;
        
        this.updateUserMarker(pos.lng, pos.lat);
        this.startWatchingPosition();
      },
      (error) => {
        console.error('Error obteniendo ubicación:', error);
        this.isLocating = false;
        alert('No se pudo obtener tu ubicación');
      },
      { enableHighAccuracy: true, timeout: 10000 }
    );
  }

  updateUserMarker(lng: number, lat: number) {
    if (!this.userMarker) {
      this.userMarker = new mapboxgl.Marker({ color: '#2563EB' })
        .setLngLat([lng, lat])
        .addTo(this.map);
    } else {
      this.userMarker.setLngLat([lng, lat]);
    }

    if (this.isFollowMode) {
      this.map.flyTo({ center: [lng, lat], zoom: 16 });
    }
  }

  startWatchingPosition() {
    this.watchId = navigator.geolocation.watchPosition(
      (position) => {
        this.userPosition = {
          lat: position.coords.latitude,
          lng: position.coords.longitude
        };
        this.updateUserMarker(position.coords.longitude, position.coords.latitude);
      },
      (error) => console.error('Error watching position:', error),
      { enableHighAccuracy: true, maximumAge: 5000 }
    );
  }

  centerUser() {
    if (this.userPosition) {
      this.map.flyTo({
        center: [this.userPosition.lng, this.userPosition.lat],
        zoom: 16
      });
    }
  }

  toggleFollowMode() {
    this.isFollowMode = !this.isFollowMode;
    if (this.isFollowMode) {
      this.centerUser();
    }
  }

  onBack() {
    this.router.navigate(['/card']);
  }
}
```
