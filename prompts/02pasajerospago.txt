# 💳 MÓDULO 02 - PASAJEROS PAGO (ScanQR + PayFare)

## 📋 Objetivo de este módulo
Implementar el **flujo de pago corregido**: El pasajero escanea el QR del microbus y paga directamente.

**⚠️ IMPORTANTE**: Este es el módulo MÁS CRÍTICO. El pasajero NO genera QR, sino que ESCANEA el QR del bus.

**Requisito previo**: Haber completado `01-PASAJEROS-CORE.md`

---

## 🔄 FLUJO CORRECTO DE PAGO

```
1. Pasajero en MyCard → Click "Escanear QR"
   ↓
2. ScanQRComponent → Escanea QR fijo del microbus
   QR: "BUS:ABC123:Ruta 5:VALID001"
   ↓
3. Sistema parsea:  placa="ABC123", ruta="Ruta 5", validador="VALID001"
   ↓
4. Navega a PayFareComponent con queryParams
   ↓
5. Pasajero selecciona adultos/escolares
   ↓
6. Click "Pagar S/ X.XX" → Pago directo (NO genera QR)
   ↓
7. WebSocket envía evento al conductor
   ↓
8. Navega a Ticket de confirmación
```

### scan-qr.component.ts
```typescript
import { Component } from '@angular/core';
import { Router } from '@angular/router';
import { BusService } from '../../../../core/services/bus.service';

@Component({
  selector: 'app-scan-qr',
  templateUrl: './scan-qr.component.html',
  styleUrls: ['./scan-qr.component.css']
})
export class ScanQRComponent {
  isScanning = false;
  availableDevices: MediaDeviceInfo[] = [];
  currentDevice: MediaDeviceInfo | undefined;
  hasPermission = false;

  constructor(
    private router: Router,
    private busService: BusService
  ) {}

  onCamerasFound(devices: MediaDeviceInfo[]): void {
    this.availableDevices = devices;
    this.hasPermission = true;
    
    // Seleccionar cámara trasera si está disponible
    const backCamera = devices.find(device => 
      /back|rear|environment/gi.test(device.label)
    );
    this.currentDevice = backCamera || devices[0];
  }

  onScanSuccess(qrData: string): void {
    try {
      // Validar y parsear QR del bus
      if (!this.busService.isValidBusQR(qrData)) {
        alert('QR inválido. Escanea el QR del microbus.');
        return;
      }

      const busData = this.busService.parseBusQR(qrData);
      
      // Navegar a PayFare con los datos del bus
      this.router.navigate(['/pay-fare'], {
        queryParams: {
          placa: busData.placa,
          ruta: busData.ruta,
          validador: busData.validador
        }
      });
    } catch (error) {
      alert('Error al escanear QR. Intenta de nuevo.');
      console.error(error);
    }
  }

  onScanError(error: Error): void {
    console.error('Error al escanear:', error);
  }

  // Para testing - simular escaneo
  simulateScan(): void {
    this.isScanning = true;
    
    setTimeout(() => {
      this.onScanSuccess('BUS:ABC123:Ruta 5:VALID001');
    }, 1500);
  }

  onBack(): void {
    this.router.navigate(['/card']);
  }
}
```

## 💰 COMPONENTE 2: PAY-FARE ⭐⭐ ACTUALIZADO (NO GENERA QR)

### pay-fare.component.ts
```typescript
import { Component, OnInit } from '@angular/core';
import { ActivatedRoute, Router } from '@angular/router';
import { BusData } from '../../../../core/models/bus-data.model';
import { TransactionService } from '../../../../core/services/transaction.service';
import { WebSocketService } from '../../../../core/services/websocket.service';
import { AuthService } from '../../../../core/services/auth.service';

@Component({
  selector: 'app-pay-fare',
  templateUrl: './pay-fare.component.html'
})
export class PayFareComponent implements OnInit {
  busData: BusData | null = null;
  adultos = 1;
  escolares = 0;
  currentBalance = 0;
  isProcessing = false;

  readonly TARIFA_ADULTO = 1.0;
  readonly TARIFA_ESCOLAR = 0.5;

  constructor(
    private route: ActivatedRoute,
    private router: Router,
    private transactionService: TransactionService,
    private wsService: WebSocketService,
    private authService: AuthService
  ) {}

  ngOnInit() {
    // Leer datos del bus desde queryParams
    this.route.queryParams.subscribe(params => {
      this.busData = {
        placa: params['placa'] || 'DEMO123',
        ruta: params['ruta'] || 'Ruta Demo',
        validador: params['validador'] || 'VALID001'
      };

      console.log('QR escaneado:', this.busData);
    });

    this.currentBalance = this.transactionService.calculateBalance();
  }

  get total(): number {
    return (this.adultos * this.TARIFA_ADULTO) + (this.escolares * this.TARIFA_ESCOLAR);
  }

  get canPay(): boolean {
    return this.total > 0 && this.total <= this.currentBalance;
  }

  incrementAdultos() {
    if (this.adultos < 10) this.adultos++;
  }

  decrementAdultos() {
    if (this.adultos > 0) this.adultos--;
  }

  incrementEscolares() {
    if (this.escolares < 10) this.escolares++;
  }

  decrementEscolares() {
    if (this.escolares > 0) this.escolares--;
  }

  onPay() {
    if (!this.canPay || !this.busData) return;

    this.isProcessing = true;

    // Simular procesamiento
    setTimeout(() => {
      // Enviar pago via WebSocket
      this.wsService.sendPayment({
        busPlaca: this.busData!.placa,
        adultos: this.adultos,
        escolares: this.escolares,
        total: this.total,
        passengerId: this.authService.getCurrentUser()?.dni || ''
      });

      // Registrar transacción local
      this.transactionService.addPayment({
        adultos: this.adultos,
        escolares: this.escolares,
        total: this.total,
        route: this.busData!.ruta,
        busPlaca: this.busData!.placa
      });

      alert('¡Pago exitoso! Puedes abordar el microbus');
      this.router.navigate(['/card']);
    }, 1000);
  }

  onBack() {
    this.router.navigate(['/scan-qr']);
  }
}
```

## 🔧 SERVICIO: BusService

### src/app/core/services/bus.service.ts
```typescript
import { Injectable } from '@angular/core';
import { BusData } from '../models/bus-data.model';

@Injectable({
  providedIn: 'root'
})
export class BusService {
  
  /**
   * Valida si un QR es del formato correcto de microbus
   * Formato esperado: "BUS:{placa}:{ruta}:{validador}"
   */
  isValidBusQR(qrData: string): boolean {
    try {
      this.parseBusQR(qrData);
      return true;
    } catch {
      return false;
    }
  }

  /**
   * Parsea el QR del microbus y extrae los datos
   * @throws Error si el formato es inválido
   */
  parseBusQR(qrData: string): BusData {
    const parts = qrData.split(':');
    
    if (parts[0] !== 'BUS') {
      throw new Error('QR no es de tipo BUS');
    }
    
    if (parts.length < 4) {
      throw new Error('Formato de QR inválido');
    }
    
    return {
      placa: parts[1],
      ruta: parts[2],
      validador: parts[3]
    };
  }

  /**
   * Genera un QR de microbus (para admin/conductor)
   * En producción esto vendría del backend
   */
  generateBusQR(placa: string, ruta: string, validador: string): string {
    return `BUS:${placa}:${ruta}:${validador}`;
  }
}
```
