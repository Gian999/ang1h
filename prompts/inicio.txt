## 🎨 ESPECIFICACIONES DE DISEÑO

### Paleta de colores:
```css
--color-blue-primary: #2563EB;    /* Pasajeros */
--color-green-secondary: #059669;  /* Acciones positivas */
--color-amber: #F59E0B;            /* Conductores */
--color-background: #F8FAFC;
--color-text: #1F2937;
--color-text-secondary: #6B7280;
```

### Reglas de tipografía:
⚠️ **IMPORTANTE**: NO usar clases de Tailwind para:
- `font-size` (text-xl, text-2xl, etc.)
- `font-weight` (font-bold, font-semibold, etc.)
- `line-height` (leading-tight, etc.)

Usar tipografía definida en `globals.css`

### Accesibilidad:
- Elementos táctiles: mínimo 44px × 44px
- Contraste: WCAG AA
- Textos: mínimo 14px

### Tono comunicacional:
- Español peruano formal pero cercano
- Tono cívico/municipal
- Términos: "saldo", "pasaje", "microbus", "DNI", "celular"

## 🔒 GUARDS

### src/app/core/guards/auth.guard.ts
```typescript
import { Injectable } from '@angular/core';
import { CanActivate, Router } from '@angular/router';
import { AuthService } from '../services/auth.service';

@Injectable({
  providedIn: 'root'
})
export class AuthGuard implements CanActivate {
  constructor(
    private authService: AuthService,
    private router: Router
  ) {}

  canActivate(): boolean {
    if (this.authService.isAuthenticated()) {
      return true;
    }
    this.router.navigate(['/login']);
    return false;
  }
}
```

### src/app/core/guards/driver.guard.ts
```typescript
import { Injectable } from '@angular/core';
import { CanActivate, Router } from '@angular/router';
import { DriverService } from '../services/driver.service';

@Injectable({
  providedIn: 'root'
})
export class DriverGuard implements CanActivate {
  constructor(
    private driverService: DriverService,
    private router: Router
  ) {}

  canActivate(): boolean {
    if (this.driverService.isAuthenticated()) {
      return true;
    }
    this.router.navigate(['/driver/login']);
    return false;
  }
}
```

## 🛣️ ROUTING PRINCIPAL

### src/app/app-routing.module.ts
```typescript
import { NgModule } from '@angular/core';
import { RouterModule, Routes } from '@angular/router';

const routes: Routes = [
  { path: '', redirectTo: '/welcome', pathMatch: 'full' },
  {
    path: '',
    loadChildren: () => import('./features/passenger/passenger.module')
      .then(m => m.PassengerModule)
  },
  {
    path: 'driver',
    loadChildren: () => import('./features/driver/driver.module')
      .then(m => m.DriverModule)
  },
  { path: '**', redirectTo: '/welcome' }
];

@NgModule({
  imports: [RouterModule.forRoot(routes)],
  exports: [RouterModule]
})
export class AppRoutingModule { }
```
