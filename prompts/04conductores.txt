```typescript
@Component({
  selector: 'app-driver-register',
  templateUrl: './driver-register.component.html'
})
export class DriverRegisterComponent {
  currentStep = 1;
  totalSteps = 4;
  
  step1Form: FormGroup; // DNI, nombres, apellidos, licencia, fechaNacimiento
  step2Form: FormGroup; // Email, celular
  step3Form: FormGroup; // ⭐ Datos del vehículo
  step4Form: FormGroup; // Password, confirmPassword, acceptTerms

  constructor(private fb: FormBuilder, private router: Router, private driverService: DriverService) {
    this.step1Form = this.fb.group({
      dni: ['', [Validators.required, Validators.pattern(/^\d{8}$/)]],
      nombres: ['', Validators.required],
      apellidos: ['', Validators.required],
      licencia: ['', [Validators.required, Validators.pattern(/^Q\d{8}$/)]],
      fechaNacimiento: ['', Validators.required]
    });

    this.step2Form = this.fb.group({
      email: ['', [Validators.required, Validators.email]],
      celular: ['', [Validators.required, Validators.pattern(/^9\d{8}$/)]]
    });

    // ⭐ Paso 3: Datos del vehículo
    this.step3Form = this.fb.group({
      placa: ['', [Validators.required, Validators.pattern(/^[A-Z0-9]{6,7}$/)]],
      modelo: ['', Validators.required],
      año: ['', [Validators.required, Validators.min(1990), Validators.max(2025)]],
      color: ['', Validators.required],
      capacidad: ['', [Validators.required, Validators.min(5), Validators.max(50)]],
      numeroRuta: ['', Validators.required],
      empresaTransporte: ['', Validators.required]
    });

    this.step4Form = this.fb.group({
      password: ['', [Validators.required, Validators.minLength(6)]],
      confirmPassword: ['', Validators.required],
      acceptTerms: [false, Validators.requiredTrue]
    });
  }

  async onSubmit() {
    const driverData = {
      ...this.step1Form.value,
      ...this.step2Form.value,
      ...this.step3Form.value,
      password: this.step4Form.value.password
    };

    const success = await this.driverService.register(driverData);
    if (success) {
      this.router.navigate(['/driver/map']);
    }
  }
}
```

## 🔐 COMPONENTE 3: DRIVER-LOGIN (2FA + GPS) ⭐

### driver-login.component.ts
```typescript
@Component({
  selector: 'app-driver-login',
  templateUrl: './driver-login.component.html'
})
export class DriverLoginComponent {
  step = 1; // 1: credentials, 2: code verification
  credentialsForm: FormGroup;
  codeForm: FormGroup;

  constructor(
    private fb: FormBuilder,
    private driverService: DriverService,
    private router: Router
  ) {
    this.credentialsForm = this.fb.group({
      identifier: ['', Validators.required],
      password: ['', Validators.required]
    });

    this.codeForm = this.fb.group({
      code: ['', [Validators.required, Validators.pattern(/^\d{6}$/)]]
    });
  }

  onSendCode() {
    if (this.credentialsForm.invalid) return;
    
    // Demo: enviar código
    alert('Código enviado a tu celular: 123456');
    this.step = 2;
  }

  async onLogin() {
    if (this.codeForm.invalid) return;

    const { identifier, password } = this.credentialsForm.value;
    const { code } = this.codeForm.value;

    // Validar credenciales demo
    if (identifier === '87654321' && password === 'conductor123' && code === '123456') {
      // ⭐ SOLICITAR PERMISOS GPS
      try {
        const permission = await navigator.permissions.query({ name: 'geolocation' as PermissionName });
        
        if (permission.state === 'denied') {
          alert('Debes permitir el acceso a la ubicación para usar la app de conductores');
          return;
        }

        navigator.geolocation.getCurrentPosition(
          () => {
            alert('¡Bienvenido conductor! Ubicación activada');
            this.driverService.login(identifier, password, code);
            this.router.navigate(['/driver/map']);
          },
          (error) => {
            if (error.code === error.PERMISSION_DENIED) {
              alert('Debes permitir el acceso a la ubicación para continuar');
            } else {
              alert('No se pudo obtener la ubicación. Intenta más tarde');
              this.router.navigate(['/driver/map']);
            }
          },
          { enableHighAccuracy: true, timeout: 10000 }
        );
      } catch (error) {
        // Fallback para navegadores sin Permissions API
        navigator.geolocation.getCurrentPosition(
          () => {
            this.driverService.login(identifier, password, code);
            this.router.navigate(['/driver/map']);
          },
          () => {
            alert('No se pudo acceder a la ubicación');
            this.router.navigate(['/driver/map']);
          }
        );
      }
    } else {
      alert('Credenciales incorrectas. Demo: DNI 87654321, Pass conductor123, Código 123456');
    }
  }
}
```

```typescript
@Component({
  selector: 'app-driver-map',
  templateUrl: './driver-map.component.html',
  styleUrls: ['./driver-map.component.css']
})
export class DriverMapComponent implements OnInit, OnDestroy, AfterViewInit {
  private map!: mapboxgl.Map;
  private busMarker?: mapboxgl.Marker;
  private watchId?: number;

  driverData: any;
  isOnDuty = false;
  isTracking = false;
  showStatsPanel = false; // ⭐ Oculto por defecto
  
  passengerCount = 0;
  totalCollected = 0;
  currentSpeed = 0;
  tripDistance = 0;
  userPosition: { lat: number; lng: number } | null = null;

  readonly MAPBOX_TOKEN = 'pk.eyJ1IjoibHVjay1wYXN0b3IiLCJhIjoiY21oOXhjczdsMDBkZjJybjFpam9oMG1qbCJ9.0Q8otsUabn_2ZQVouvfiJw';

  constructor(
    private driverService: DriverService,
    private wsService: WebSocketService,
    private router: Router
  ) {}

  ngOnInit() {
    this.driverData = this.driverService.getCurrentDriver();
    
    // Conectar WebSocket
    this.wsService.connect();
    this.wsService.driverConnect({
      placa: this.driverData.placa,
      ruta: this.driverData.numeroRuta,
      driverName: `${this.driverData.nombres} ${this.driverData.apellidos}`
    });

    // Escuchar pasajeros
    this.wsService.onPassengerBoarded().subscribe(event => {
      if (event) {
        this.passengerCount = event.totalPassengers;
        this.totalCollected = event.totalCollected;
        
        // ⭐ Expandir panel automáticamente
        if (!this.showStatsPanel && this.isOnDuty) {
          this.showStatsPanel = true;
        }

        alert(`${event.adultos + event.escolares} pasajero(s) - S/ ${event.total.toFixed(2)}`);
      }
    });
  }

  ngAfterViewInit() {
    this.initMap();
  }

  ngOnDestroy() {
    if (this.watchId) {
      navigator.geolocation.clearWatch(this.watchId);
    }
    this.wsService.disconnect();
  }

  initMap() {
    mapboxgl.accessToken = this.MAPBOX_TOKEN;

    this.map = new mapboxgl.Map({
      container: 'driver-map',
      style: 'mapbox://styles/mapbox/streets-v12',
      center: [-70.13, -15.5],
      zoom: 14,
      pitch: 45, // Vista 3D opcional
      bearing: 0
    });
  }

  handleToggleDuty(checked: boolean) {
    this.isOnDuty = checked;
    
    if (checked) {
      alert('¡En servicio! Tracking GPS iniciado');
      this.startTracking();
    } else {
      alert('Fuera de servicio. Tracking detenido');
      this.stopTracking();
      this.passengerCount = 0;
      this.totalCollected = 0;
      this.tripDistance = 0;
    }
  }

  startTracking() {
    this.isTracking = true;
    
    this.watchId = navigator.geolocation.watchPosition(
      (position) => {
        this.userPosition = {
          lat: position.coords.latitude,
          lng: position.coords.longitude
        };
        
        if (position.coords.speed) {
          this.currentSpeed = Math.round(position.coords.speed * 3.6);
        }

        this.updateBusMarker();
      },
      (error) => console.error('Error tracking:', error),
      { enableHighAccuracy: true, maximumAge: 1000, timeout: 5000 }
    );
  }

  stopTracking() {
    if (this.watchId) {
      navigator.geolocation.clearWatch(this.watchId);
      this.watchId = undefined;
    }
    this.isTracking = false;
  }

  updateBusMarker() {
    if (!this.userPosition) return;

    if (!this.busMarker) {
      this.busMarker = new mapboxgl.Marker({ color: '#F59E0B' })
        .setLngLat([this.userPosition.lng, this.userPosition.lat])
        .addTo(this.map);
    } else {
      this.busMarker.setLngLat([this.userPosition.lng, this.userPosition.lat]);
    }

    this.map.flyTo({ center: [this.userPosition.lng, this.userPosition.lat] });
  }

  toggleStatsPanel() {
    this.showStatsPanel = !this.showStatsPanel;
  }

  // Demo
  simulatePassengerScan() {
    this.wsService.sendPayment({
      busPlaca: this.driverData.placa,
      adultos: 1,
      escolares: 0,
      total: 1.0,
      passengerId: 'DEMO'
    });
  }
}
```
